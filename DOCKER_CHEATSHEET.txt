┌────────────────────────────────────────────────────────────────┐
│ ⚡ COMANDOS PRINCIPALES                                        │
└────────────────────────────────────────────────────────────────┘

🟢 INICIAR
──────────────────────────────────────────────────────────────────
  Producción (Nginx):
    docker-compose up -d
    → http://localhost

  Desarrollo (ng serve):
    docker-compose -f docker-compose.dev.yml up
    → http://localhost:4200 (con hot reload)

🔴 DETENER
──────────────────────────────────────────────────────────────────
  docker-compose down              # Detener
  docker-compose down -v           # Detener + eliminar volúmenes

🔄 REINICIAR
──────────────────────────────────────────────────────────────────
  docker-compose restart           # Reiniciar
  docker-compose restart qams-web  # Reiniciar servicio específico

📊 VER ESTADO
──────────────────────────────────────────────────────────────────
  docker-compose ps                # Estado contenedores
  docker ps                        # Estado (docker directo)
  docker images | grep qams        # Listar imágenes

📝 LOGS
──────────────────────────────────────────────────────────────────
  docker-compose logs -f qams-web              # Logs en tiempo real
  docker-compose logs --tail=100 qams-web      # Últimas 100 líneas
  docker logs -f qams-web                      # Logs (docker directo)

🔨 CONSTRUIR
──────────────────────────────────────────────────────────────────
  docker-compose build             # Construir con caché
  docker-compose build --no-cache  # Construir sin caché
  docker build -t qams-web:latest .            # Docker directo

💻 SHELL
──────────────────────────────────────────────────────────────────
  docker-compose exec qams-web sh  # Abrir shell en contenedor
  docker exec -it qams-web sh      # Docker directo

🪟 WINDOWS (docker.bat)
──────────────────────────────────────────────────────────────────
  docker.bat start         # Iniciar
  docker.bat stop          # Detener
  docker.bat restart       # Reiniciar
  docker.bat logs          # Ver logs
  docker.bat build         # Construir
  docker.bat rebuild       # Reconstruir (sin caché)
  docker.bat ps            # Ver estado
  docker.bat shell         # Abrir shell
  docker.bat clean         # Limpiar todo

🐧 LINUX/MAC (docker.sh)
──────────────────────────────────────────────────────────────────
  ./docker.sh start        # Iniciar
  ./docker.sh stop         # Detener
  ./docker.sh restart      # Reiniciar
  ./docker.sh logs         # Ver logs
  ./docker.sh build        # Construir
  ./docker.sh rebuild      # Reconstruir (sin caché)
  ./docker.sh ps           # Ver estado
  ./docker.sh shell        # Abrir shell
  ./docker.sh clean        # Limpiar todo

──────────────────────────────────────────────────────────────────
  docker run -d -p 80:80 --name qams-web qams-web:latest

──────────────────────────────────────────────────────────────────
  docker stop qams-web             # Detener
  docker start qams-web            # Iniciar
  docker rm qams-web               # Eliminar
  docker rmi qams-web:latest       # Eliminar imagen

ESTADÍSTICAS
──────────────────────────────────────────────────────────────────
  docker stats qams-web            # Uso de recursos
  docker inspect qams-web          # Información detallada
  docker logs -f qams-web          # Logs

┌────────────────────────────────────────────────────────────────┐
│ 🔧 DOCKER-COMPOSE AVANZADO                                     │
└────────────────────────────────────────────────────────────────┘

MÚLTIPLES SERVICIOS
──────────────────────────────────────────────────────────────────
  # Producción completa (Web + API + DB + Redis)
  docker-compose -f docker-compose.prod.yml up -d

  # Ver estado
  docker-compose -f docker-compose.prod.yml ps

  # Logs de servicio específico
  docker-compose -f docker-compose.prod.yml logs -f qams-api

VARIABLES DE AMBIENTE
──────────────────────────────────────────────────────────────────
  # En línea de comandos
  API_URL=http://api:3000 docker-compose up -d

  # O desde archivo .env
  export NODE_ENV=production
  docker-compose up -d

VOLÚMENES
──────────────────────────────────────────────────────────────────
  docker volume ls                 # Listar volúmenes
  docker volume inspect db-data    # Inspeccionar
  docker volume prune              # Limpiar no utilizados
  docker-compose down -v           # Eliminar con compose

REDES
──────────────────────────────────────────────────────────────────
  docker network ls                # Listar redes
  docker network inspect qams_network  # Ver detalles

  # Volumen
  docker volume rm qams-web_db-data

┌────────────────────────────────────────────────────────────────┐
│ 🚨 TROUBLESHOOTING                                             │
└────────────────────────────────────────────────────────────────┘

PROBLEMA: Puerto 80 ya está en uso
──────────────────────────────────────────────────────────────────
Solución 1: Cambiar puerto en docker-compose.yml
  ports:
    - "8080:80"  # Cambiar 80 por otro puerto

Solución 2: Ver qué usa el puerto
  # Windows
  netstat -ano | findstr :80

  # Linux/Mac
  lsof -i :80

PROBLEMA: Contenedor no inicia
──────────────────────────────────────────────────────────────────
  docker-compose logs qams-web
  docker logs qams-web

PROBLEMA: Cambios de código no se ven
──────────────────────────────────────────────────────────────────
  # Usar versión de desarrollo con hot reload
  docker-compose -f docker-compose.dev.yml up

PROBLEMA: Permisos denegados
──────────────────────────────────────────────────────────────────
  # Linux/Mac
  chmod +x docker.sh
  sudo docker-compose ...  # Si es necesario

PROBLEMA: Caché antiguo
──────────────────────────────────────────────────────────────────
  docker-compose build --no-cache
  docker system prune -a
